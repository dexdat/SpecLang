```typescript
// src/safety/fallback.ts

export type AgentSupportLevel = 'human_only' | 'agent_assisted' | 'agent_autonomous';

export interface FallbackAction {
  type: 'proceed' | 'warn' | 'review' | 'quarantine' | 'downgrade';
  spec: Spec;
  previousLevel: AgentSupportLevel;
  newLevel: AgentSupportLevel;
  reason: string;
  details: {
    confidenceScore?: number;
    threshold?: number;
    detectedIssues?: string[];
    ticketId?: string;
    reviewers?: string[];
  };
  timestamp: string;
  autoGenerated: boolean;
}

export interface FallbackConfig {
  thresholds: {
    quarantine: number;      // Below this = quarantine
    review: number;          // Below this = require review
    warn: number;             // Below this = warn only
    requireHumanReview: number; // Below this = downgrade to human_only
  };
  downgradeSteps: {
    from: AgentSupportLevel;
    to: AgentSupportLevel;
  }[];
  autoDowngrade: boolean;
  requireApproval: {
    toQuarantine: boolean;
    toReview: boolean;
    toDowngrade: boolean;
  };
  cooldownMinutes: number;    // Cooldown between fallbacks
  maxQuarantineDuration: number; // Hours before auto-release
}

export class FallbackProtocol {
  private config: FallbackConfig;
  private confidenceScorer: ConfidenceScorer;
  private quarantineManager: QuarantineManager;
  private peerReview: PeerReviewWorkflow;
  private notification: NotificationService;
  private cooldown: Map<string, number> = new Map();
  
  constructor(
    confidenceScorer: ConfidenceScorer,
    config: Partial<FallbackConfig> = {}
  ) {
    this.confidenceScorer = confidenceScorer;
    this.quarantineManager = new QuarantineManager();
    this.peerReview = new PeerReviewWorkflow();
    this.notification = new NotificationService();
    this.config = this.mergeConfig(config);
  }

  async evaluate(spec: Spec): Promise<FallbackAction> {
    // Check cooldown
    if (this.isInCooldown(spec.id)) {
      return this.createProceedAction(spec);
    }
    
    // Get confidence score
    const score = await this.confidenceScorer.score(spec);
    const currentLevel = spec.metadata.agent_support as AgentSupportLevel;
    
    // Determine action based on score
    let action: FallbackAction;
    
    if (score.score < this.config.thresholds.quarantine) {
      action = await this.quarantine(spec, score, currentLevel);
    } else if (score.score < this.config.thresholds.review) {
      action = await this.requestReview(spec, score, currentLevel);
    } else if (score.score < this.config.thresholds.warn) {
      action = this.warn(spec, score, currentLevel);
    } else if (score.score < this.config.thresholds.requireHumanReview && 
               currentLevel !== 'human_only') {
      action = await this.downgrade(spec, score, currentLevel);
    } else {
      action = this.createProceedAction(spec);
    }
    
    // Set cooldown if action taken
    if (action.type !== 'proceed') {
      this.setCooldown(spec.id);
    }
    
    // Send notifications
    await this.sendNotifications(action);
    
    return action;
  }

  private async quarantine(
    spec: Spec,
    score: ConfidenceScore,
    currentLevel: AgentSupportLevel
  ): Promise<FallbackAction> {
    // Downgrade to human_only
    spec.metadata.agent_support = 'human_only';
    
    // Add quarantine tag
    spec.metadata.tags = [
      ...(spec.metadata.tags || []),
      'quarantined',
      `confidence:${score.score.toFixed(2)}`,
    ];
    
    // Create review ticket
    const ticket = await this.quarantineManager.createTicket({
      specId: spec.id,
      reason: `Confidence score ${score.score.toFixed(2)} below threshold ${this.config.thresholds.quarantine}`,
      severity: 'high',
      autoGenerated: true,
    });
    
    // Find reviewers
    const reviewers = await this.findReviewers(spec);
    
    return {
      type: 'quarantine',
      spec,
      previousLevel: currentLevel,
      newLevel: 'human_only',
      reason: `Confidence score ${score.score.toFixed(2)} is below quarantine threshold ${this.config.thresholds.quarantine}`,
      details: {
        confidenceScore: score.score,
        threshold: this.config.thresholds.quarantine,
        detectedIssues: this.getDetectedIssues(score),
        ticketId: ticket.id,
        reviewers,
      },
      timestamp: new Date().toISOString(),
      autoGenerated: true,
    };
  }

  private async requestReview(
    spec: Spec,
    score: ConfidenceScore,
    currentLevel: AgentSupportLevel
  ): Promise<FallbackAction> {
    // Add review tag
    spec.metadata.tags = [
      ...(spec.metadata.tags || []),
      'needs_review',
      `confidence:${score.score.toFixed(2)}`,
    ];
    
    // Create review request
    const reviewRequest = await this.peerReview.createRequest({
      specId: spec.id,
      type: 'confidence',
      priority: score.score < this.config.thresholds.review ? 'high' : 'medium',
      assignees: await this.findReviewers(spec),
    });
    
    return {
      type: 'review',
      spec,
      previousLevel: currentLevel,
      newLevel: currentLevel, // Stay at current level pending review
      reason: `Confidence score ${score.score.toFixed(2)} requires human review`,
      details: {
        confidenceScore: score.score,
        threshold: this.config.thresholds.review,
        detectedIssues: this.getDetectedIssues(score),
        ticketId: reviewRequest.id,
        reviewers: reviewRequest.assignees,
      },
      timestamp: new Date().toISOString(),
      autoGenerated: true,
    };
  }

  private warn(
    spec: Spec,
    score: ConfidenceScore,
    currentLevel: AgentSupportLevel
  ): FallbackAction {
    // Add warning tag but don't change level
    spec.metadata.tags = [
      ...(spec.metadata.tags || []),
      'low_confidence',
      `confidence:${score.score.toFixed(2)}`,
    ];
    
    return {
      type: 'warn',
      spec,
      previousLevel: currentLevel,
      newLevel: currentLevel,
      reason: `Confidence score ${score.score.toFixed(2)} is below warning threshold`,
      details: {
        confidenceScore: score.score,
        threshold: this.config.thresholds.warn,
        detectedIssues: this.getDetectedIssues(score),
      },
      timestamp: new Date().toISOString(),
      autoGenerated: true,
    };
  }

  private async downgrade(
    spec: Spec,
    score: ConfidenceScore,
    currentLevel: AgentSupportLevel
  ): Promise<FallbackAction> {
    const newLevel = this.getDowngradeLevel(currentLevel);
    
    spec.metadata.agent_support = newLevel;
    spec.metadata.tags = [
      ...(spec.metadata.tags || []),
      'downgraded',
      `confidence:${score.score.toFixed(2)}`,
    ];
    
    return {
      type: 'downgrade',
      spec,
      previousLevel: currentLevel,
      newLevel,
      reason: `Confidence score ${score.score.toFixed(2)} requires downgrade from ${currentLevel} to ${newLevel}`,
      details: {
        confidenceScore: score.score,
        threshold: this.config.thresholds.requireHumanReview,
        detectedIssues: this.getDetectedIssues(score),
      },
      timestamp: new Date().toISOString(),
      autoGenerated: true,
    };
  }

  private createProceedAction(spec: Spec): FallbackAction {
    return {
      type: 'proceed',
      spec,
      previousLevel: spec.metadata.agent_support as AgentSupportLevel,
      newLevel: spec.metadata.agent_support as AgentSupportLevel,
      reason: 'Confidence score acceptable',
      details: {},
      timestamp: new Date().toISOString(),
      autoGenerated: false,
    };
  }

  private getDowngradeLevel(current: AgentSupportLevel): AgentSupportLevel {
    const downgradeMap: Record<AgentSupportLevel, AgentSupportLevel> = {
      'agent_autonomous': 'agent_assisted',
      'agent_assisted': 'human_only',
      'human_only': 'human_only',
    };
    
    return downgradeMap[current];
  }

  private getDetectedIssues(score: ConfidenceScore): string[] {
    const issues: string[] = [];
    
    if (score.factors.historicalReliability < 0.5) {
      issues.push('Low historical reliability');
    }
    if (score.factors.authorReputation < 0.5) {
      issues.push('Unknown or low-reputation author');
    }
    if (score.factors.similarityToGood < 0.5) {
      issues.push('Low similarity to known-good specs');
    }
    if (score.factors.testCoverage < 0.5) {
      issues.push('Low test coverage');
    }
    if (score.factors.reviewStatus < 0.5) {
      issues.push('Not reviewed by humans');
    }
    
    return issues;
  }

  private async findReviewers(spec: Spec): Promise<string[]> {
    // Find reviewers based on expertise matching spec tags
    const tags = spec.metadata.tags || [];
    
    // Simple implementation: return configured reviewers
    const allReviewers = ['reviewer1', 'reviewer2'];
    
    return allReviewers.slice(0, 2);
  }

  private async sendNotifications(action: FallbackAction): Promise<void> {
    switch (action.type) {
      case 'quarantine':
        await this.notification.send({
          type: 'quarantine',
          specId: action.spec.id,
          message: action.reason,
          ticketId: action.details.ticketId,
        });
        break;
        
      case 'review':
        await this.notification.send({
          type: 'review_request',
          specId: action.spec.id,
          message: action.reason,
          assignees: action.details.reviewers,
        });
        break;
        
      case 'downgrade':
        await this.notification.send({
          type: 'downgrade',
          specId: action.spec.id,
          message: action.reason,
          previousLevel: action.previousLevel,
          newLevel: action.newLevel,
        });
        break;
        
      case 'warn':
        await this.notification.send({
          type: 'warning',
          specId: action.spec.id,
          message: action.reason,
        });
        break;
    }
  }

  private isInCooldown(specId: string): boolean {
    const lastAction = this.cooldown.get(specId);
    if (!lastAction) return false;
    
    const cooldownMs = this.config.cooldownMinutes * 60 * 1000;
    return Date.now() - lastAction < cooldownMs;
  }

  private setCooldown(specId: string): void {
    this.cooldown.set(specId, Date.now());
  }

  private mergeConfig(config: Partial<FallbackConfig>): FallbackConfig {
    return {
      thresholds: { ...defaultConfig.thresholds, ...config.thresholds },
      downgradeSteps: config.downgradeSteps || defaultConfig.downgradeSteps,
      autoDowngrade: config.autoDowngrade ?? defaultConfig.autoDowngrade,
      requireApproval: { ...defaultConfig.requireApproval, ...config.requireApproval },
      cooldownMinutes: config.cooldownMinutes ?? defaultConfig.cooldownMinutes,
      maxQuarantineDuration: config.maxQuarantineDuration ?? defaultConfig.maxQuarantineDuration,
    };
  }

  // Manual override methods
  async approveQuarantine(specId: string, approver: string): Promise<void> {
    await this.quarantineManager.approve(specId, approver);
  }

  async releaseFromQuarantine(specId: string, approver: string): Promise<void> {
    await this.quarantineManager.release(specId, approver);
  }

  async restoreLevel(specId: string, newLevel: AgentSupportLevel): Promise<void> {
    // Restore spec to higher autonomy level
    // Should only be called after human review approves
  }
}

const defaultConfig: FallbackConfig = {
  thresholds: {
    quarantine: 0.5,
    review: 0.6,
    warn: 0.7,
    requireHumanReview: 0.65,
  },
  downgradeSteps: [
    { from: 'agent_autonomous', to: 'agent_assisted' },
    { from: 'agent_assisted', to: 'human_only' },
  ],
  autoDowngrade: true,
  requireApproval: {
    toQuarantine: true,
    toReview: false,
    toDowngrade: false,
  },
  cooldownMinutes: 60,
  maxQuarantineDuration: 24,
};
```

```typescript
export interface QuarantineTicket {
  id: string;
  specId: string;
  reason: string;
  severity: 'low' | 'medium' | 'high' | 'critical';
  status: 'open' | 'approved' | 'released' | 'expired';
  createdAt: string;
  expiresAt: string;
  approvedBy: string | null;
  approvedAt: string | null;
  releaseReason: string | null;
  autoGenerated: boolean;
}

export class QuarantineManager {
  private tickets: Map<string, QuarantineTicket> = new Map();
  private db: Database;
  
  constructor() {
    this.db = new Database('safety/quarantine.db');
  }

  async createTicket(data: {
    specId: string;
    reason: string;
    severity: QuarantineTicket['severity'];
    autoGenerated: boolean;
  }): Promise<QuarantineTicket> {
    const ticket: QuarantineTicket = {
      id: this.generateId(),
      specId: data.specId,
      reason: data.reason,
      severity: data.severity,
      status: 'open',
      createdAt: new Date().toISOString(),
      expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(), // 24 hours
      approvedBy: null,
      approvedAt: null,
      releaseReason: null,
      autoGenerated: data.autoGenerated,
    };
    
    this.tickets.set(ticket.id, ticket);
    
    // Persist to database
    await this.db.run(`
      INSERT INTO quarantine_tickets (id, spec_id, reason, severity, status, created_at, expires_at, auto_generated)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `, ticket.id, ticket.specId, ticket.reason, ticket.severity, ticket.status, ticket.createdAt, ticket.expiresAt, ticket.autoGenerated);
    
    return ticket;
  }

  async approve(ticketId: string, approver: string): Promise<void> {
    const ticket = this.tickets.get(ticketId);
    if (!ticket) throw new Error(`Ticket ${ticketId} not found`);
    
    ticket.status = 'approved';
    ticket.approvedBy = approver;
    ticket.approvedAt = new Date().toISOString();
    
    await this.db.run(`
      UPDATE quarantine_tickets SET status = ?, approved_by = ?, approved_at = ? WHERE id = ?
    `, ticket.status, ticket.approvedBy, ticket.approvedAt, ticket.id);
  }

  async release(ticketId: string, releaser: string, reason?: string): Promise<void> {
    const ticket = this.tickets.get(ticketId);
    if (!ticket) throw new Error(`Ticket ${ticketId} not found`);
    
    ticket.status = 'released';
    ticket.releaseReason = reason || 'Manually released';
    
    await this.db.run(`
      UPDATE quarantine_tickets SET status = ?, release_reason = ? WHERE id = ?
    `, ticket.status, ticket.releaseReason, ticket.id);
  }

  async getOpenTickets(): Promise<QuarantineTicket[]> {
    return Array.from(this.tickets.values()).filter(t => t.status === 'open');
  }

  async getTicketsForSpec(specId: string): Promise<QuarantineTicket[]> {
    return Array.from(this.tickets.values()).filter(t => t.specId === specId);
  }

  private generateId(): string {
    return `QT-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  }
}
```

```typescript
export interface ReviewRequest {
  id: string;
  specId: string;
  type: 'confidence' | 'manual' | 'critical_change';
  priority: 'low' | 'medium' | 'high' | 'critical';
  status: 'pending' | 'in_progress' | 'approved' | 'rejected' | 'changes_requested';
  assignees: string[];
  createdAt: string;
  dueAt: string | null;
  completedAt: string | null;
  comments: ReviewComment[];
}

export interface ReviewComment {
  id: string;
  author: string;
  content: string;
  createdAt: string;
  resolved: boolean;
}

export class PeerReviewWorkflow {
  private requests: Map<string, ReviewRequest> = new Map();
  
  async createRequest(data: {
    specId: string;
    type: ReviewRequest['type'];
    priority: ReviewRequest['priority'];
    assignees: string[];
  }): Promise<ReviewRequest> {
    const request: ReviewRequest = {
      id: this.generateId(),
      specId: data.specId,
      type: data.type,
      priority: data.priority,
      status: 'pending',
      assignees: data.assignees,
      createdAt: new Date().toISOString(),
      dueAt: this.calculateDueDate(data.priority),
      completedAt: null,
      comments: [],
    };
    
    this.requests.set(request.id, request);
    
    // Notify assignees
    await this.notifyAssignees(request);
    
    return request;
  }

  async addComment(
    requestId: string,
    author: string,
    content: string
  ): Promise<ReviewComment> {
    const request = this.requests.get(requestId);
    if (!request) throw new Error(`Request ${requestId} not found`);
    
    const comment: ReviewComment = {
      id: this.generateId(),
      author,
      content,
      createdAt: new Date().toISOString(),
      resolved: false,
    };
    
    request.comments.push(comment);
    
    return comment;
  }

  async approve(requestId: string, reviewer: string): Promise<void> {
    const request = this.requests.get(requestId);
    if (!request) throw new Error(`Request ${requestId} not found`);
    
    // Verify reviewer is assigned
    if (!request.assignees.includes(reviewer)) {
      throw new Error(`Reviewer ${reviewer} not assigned to this request`);
    }
    
    request.status = 'approved';
    request.completedAt = new Date().toISOString();
    
    // Update spec tags
    await this.updateSpecAfterApproval(request.specId);
  }

  async reject(requestId: string, reviewer: string, reason: string): Promise<void> {
    const request = this.requests.get(requestId);
    if (!request) throw new Error(`Request ${requestId} not found`);
    
    request.status = 'rejected';
    request.completedAt = new Date().toISOString();
    
    await this.addComment(requestId, reviewer, `Rejected: ${reason}`);
  }

  async requestChanges(requestId: string, reviewer: string, feedback: string): Promise<void> {
    const request = this.requests.get(requestId);
    if (!request) throw new Error(`Request ${requestId} not found`);
    
    request.status = 'changes_requested';
    
    await this.addComment(requestId, reviewer, `Changes requested: ${feedback}`);
  }

  async getPendingForReviewer(reviewer: string): Promise<ReviewRequest[]> {
    return Array.from(this.requests.values()).filter(
      r => r.assignees.includes(reviewer) && r.status === 'pending'
    );
  }

  private calculateDueDate(priority: ReviewRequest['priority']): string | null {
    const hours: Record<ReviewRequest['priority'], number> = {
      critical: 4,
      high: 24,
      medium: 72,
      low: 168, // 1 week
    };
    
    return new Date(Date.now() + hours[priority] * 60 * 60 * 1000).toISOString();
  }

  }
```

```typescript
// src/safety/notification.ts

export interface Notification {
  type: 'quarantine' | 'review_request' | 'downgrade' | 'warning' | 'approval';
  specId: string;
  message: string;
  ticketId?: string;
  assignees?: string[];
  previousLevel?: string;
  newLevel?: string;
}

export class NotificationService {
  async send(notification: Notification): Promise<void> {
    // Format message based on type
    const formatted = this.formatNotification(notification);
    
    // Log notification (in production, integrate with actual notification systems)
    console.log(`[NOTIFICATION] ${notification.type}: ${formatted}`);
    
    // Store notification
    await this.store(notification);
  }

  private formatNotification(notification: Notification): string {
    switch (notification.type) {
      case 'quarantine':
        return `Spec ${notification.specId} quarantined. ${notification.message}. Ticket: ${notification.ticketId}`;
        
      case 'review_request':
        return `Review requested for ${notification.specId}. Assignees: ${notification.assignees?.join(', ')}`;
        
      case 'downgrade':
        return `Spec ${notification.specId} downgraded from ${notification.previousLevel} to ${notification.newLevel}`;
        
      case 'warning':
        return `Warning for ${notification.specId}: ${notification.message}`;
        
      case 'approval':
        return `Spec ${notification.specId} approved`;
        
      default:
        return notification.message;
    }
  }

  private async store(notification: Notification): Promise<void> {
    // Store in database for audit trail
  }
}
```
