#### 1. Go Type Mapping (go/types.ts)
```typescript
interface GoTypeMapping {
  stdlib: string;
  go: string;
  import?: string;        // Required import
  zeroValue: string;      // Go zero value
}

const GO_TYPE_MAPPINGS: GoTypeMapping[] = [
  // Primitives
  { stdlib: 'String', go: 'string', zeroValue: '""' },
  { stdlib: 'Int', go: 'int', zeroValue: '0' },
  { stdlib: 'Int32', go: 'int32', zeroValue: '0' },
  { stdlib: 'Int64', go: 'int64', zeroValue: '0' },
  { stdlib: 'UInt', go: 'uint', zeroValue: '0' },
  { stdlib: 'Float', go: 'float64', zeroValue: '0.0' },
  { stdlib: 'Float32', go: 'float32', zeroValue: '0.0' },
  { stdlib: 'Bool', go: 'bool', zeroValue: 'false' },
  
  // Time
  { stdlib: 'Date', go: 'time.Time', import: 'time', zeroValue: 'time.Time{}' },
  { stdlib: 'DateTime', go: 'time.Time', import: 'time', zeroValue: 'time.Time{}' },
  
  // Identifiers
  { stdlib: 'UUID', go: 'uuid.UUID', import: 'github.com/google/uuid', zeroValue: 'uuid.Nil' },
  
  // Collections
  { stdlib: 'Array<T>', go: '[]T', zeroValue: 'nil' },
  { stdlib: 'Map<K,V>', go: 'map[K]V', zeroValue: 'nil' },
  
  // Optional
  { stdlib: 'Optional<T>', go: '*T', zeroValue: 'nil' },
  
  // Bytes
  { stdlib: 'Bytes', go: '[]byte', zeroValue: 'nil' },
];

function mapGoType(stdlibType: string): { type: string; imports: string[] } {
  // Handle generics: Array<User> -> []User
  const arrayMatch = stdlibType.match(/^Array<(.+)>$/);
  if (arrayMatch) {
    const inner = mapGoType(arrayMatch[1]);
    return { type: `[]${inner.type}`, imports: inner.imports };
  }
  
  // Handle maps: Map<string, User> -> map[string]User
  const mapMatch = stdlibType.match(/^Map<(.+),\s*(.+)>$/);
  if (mapMatch) {
    const key = mapGoType(mapMatch[1]);
    const value = mapGoType(mapMatch[2]);
    return { 
      type: `map[${key.type}]${value.type}`, 
      imports: [...key.imports, ...value.imports] 
    };
  }
  
  // Handle optional: Optional<User> -> *User
  const optMatch = stdlibType.match(/^Optional<(.+)>$/);
  if (optMatch) {
    const inner = mapGoType(optMatch[1]);
    return { type: `*${inner.type}`, imports: inner.imports };
  }
  
  // Lookup base type
  const mapping = GO_TYPE_MAPPINGS.find(m => m.stdlib === stdlibType);
  if (mapping) {
    return { 
      type: mapping.go, 
      imports: mapping.import ? [mapping.import] : [] 
    };
  }
  
  // Unknown type - pass through
  return { type: stdlibType, imports: [] };
}
```

#### 2. Go Naming Conventions (go/naming.ts)
```typescript
class GoNaming {
  // Convert to PascalCase for exported types
  toTypeName(name: string): string {
    return this.toPascalCase(name);
  }
  
  // Convert to camelCase for functions/methods
  toFuncName(name: string): string {
    return this.toCamelCase(name);
  }
  
  // Convert to SCREAMING_SNAKE_CASE for constants
  toConstName(name: string): string {
    return name.replace(/([a-z])([A-Z])/g, '$1_$2').toUpperCase();
  }
  
  // Package name (lowercase, no underscores preferred)
  toPackageName(path: string): string {
    const parts = path.split('/');
    const last = parts[parts.length - 1];
    return last.toLowerCase().replace(/[^a-z0-9]/g, '');
  }
  
  private toPascalCase(s: string): string {
    return s.replace(/(^|[-_\s]+)(.)/g, (_, __, c) => c.toUpperCase());
  }
  
  private toCamelCase(s: string): string {
    const pascal = this.toPascalCase(s);
    return pascal[0].toLowerCase() + pascal.slice(1);
  }
}
```

#### 4. Go Import Handling (go/imports.ts)
```typescript
interface GoImport {
  alias?: string;
  path: string;
}

class GoImportManager {
  private imports: Map<string, GoImport> = new Map();
  
  add(pkg: string, alias?: string): void {
    if (!this.imports.has(pkg)) {
      this.imports.set(pkg, { path: pkg, alias });
    }
  }
  
  addStandard(pkg: string): void {
    this.add(pkg);
  }
  
  addThirdParty(path: string, alias?: string): void {
    this.add(path, alias);
  }
  
  render(): string {
    if (this.imports.size === 0) return '';
    
    const standard: string[] = [];
    const thirdParty: string[] = [];
    
    // Categorize imports
    for (const [path, imp] of this.imports) {
      if (this.isStandard(path)) {
        standard.push(this.formatImport(imp));
      } else {
        thirdParty.push(this.formatImport(imp));
      }
    }
    
    // Format with grouping
    const groups: string[] = [];
    
    if (standard.length > 0) {
      groups.push(standard.sort().join('\n'));
    }
    
    if (thirdParty.length > 0) {
      groups.push(thirdParty.sort().join('\n'));
    }
    
    if (groups.length === 1 && standard.length + thirdParty.length < 3) {
      // Single line import
      const all = [...standard, ...thirdParty];
      return `import ${all[0]}\n`;
    }
    
    // Multi-line import block
    return `import (
${groups.join('\n\n')}
)
`;
  }
  
  private isStandard(path: string): boolean {
    // Go stdlib packages
    const stdlibPrefixes = [
      'archive', 'bufio', 'bytes', 'compress', 'container',
      'context', 'crypto', 'database', 'debug', 'embed',
      'encoding', 'errors', 'expvar', 'flag', 'fmt',
      'go', 'hash', 'html', 'image', 'index', 'io',
      'log', 'map', 'math', 'mime', 'net', 'os',
      'path', 'plugin', 'reflect', 'regexp', 'runtime',
      'sort', 'strconv', 'strings', 'sync', 'syscall',
      'testing', 'text', 'time', 'unicode', 'unsafe'
    ];
    return stdlibPrefixes.some(p => path === p || path.startsWith(p + '/'));
  }
  
  private formatImport(imp: GoImport): string {
    if (imp.alias) {
      return `${imp.alias} "${imp.path}"`;
    }
    return `"${imp.path}"`;
  }
}
```

#### 5. Go Generator (targets/go.ts)
```typescript
class GoGenerator implements TargetGenerator {
  language = 'go';
  extension = '.go';
  
  private naming = new GoNaming();
  private importManager: GoImportManager;
  
  generate(spec: CodeSpec): GeneratedFile[] {
    const files: GeneratedFile[] = [];
    
    // Determine package from output path
    const packageName = this.naming.toPackageName(spec.target.outputPath);
    
    for (const block of spec.blocks) {
      switch (block.kind) {
        case 'struct':
        case 'entity':
          files.push(this.generateStruct(spec, block, packageName));
          break;
        case 'interface':
          files.push(this.generateInterface(spec, block, packageName));
          break;
        case 'function':
        case 'operation':
          files.push(this.generateFunction(spec, block, packageName));
          break;
        case 'enum':
          files.push(this.generateEnum(spec, block, packageName));
          break;
      }
    }
    
    return this.combineFiles(files, spec.target.outputPath);
  }
  
  private generateStruct(
    spec: CodeSpec, 
    block: CodeBlock, 
    packageName: string
  ): GeneratedFile {
    this.importManager = new GoImportManager();
    
    const structName = this.naming.toTypeName(block.name);
    const fields = this.parseStructFields(block.content);
    
    const fieldLines = fields.map(f => {
      const { type, imports } = mapGoType(f.type);
      imports.forEach(i => this.importManager.add(i));
      
      const jsonTag = f.jsonName 
        ? `\`json:"${f.jsonName}${f.optional ? ',omitempty' : ''}"\``
        : '';
      
      return `    ${this.naming.toPascalCase(f.name)} ${type} ${jsonTag}`;
    }).join('\n');
    
    const structCode = `type ${structName} struct {
${fieldLines}
}`;
    
    return {
      path: spec.target.outputPath,
      content: this.wrapInFile(structCode, spec, packageName),
      source_block: block.id
    };
  }
  
  private generateFunction(
    spec: CodeSpec, 
    block: CodeBlock, 
    packageName: string
  ): GeneratedFile {
    this.importManager = new GoImportManager();
    
    const funcDef = this.parseFunctionDef(block.content);
    const funcName = this.naming.toFuncName(funcDef.name);
    
    // Map parameter types
    const params = funcDef.params.map(p => {
      const { type, imports } = mapGoType(p.type);
      imports.forEach(i => this.importManager.add(i));
      return `${p.name} ${type}`;
    }).join(', ');
    
    // Map return type
    const { type: returnType, imports: retImports } = mapGoType(funcDef.returnType);
    retImports.forEach(i => this.importManager.add(i));
    
    const funcCode = `func ${funcName}(${params}) ${returnType} {
${this.generateFunctionBody(funcDef, block)}
}`;
    
    return {
      path: spec.target.outputPath,
      content: this.wrapInFile(funcCode, spec, packageName),
      source_block: block.id
    };
  }
  
  private wrapInFile(
    content: string, 
    spec: CodeSpec, 
    packageName: string
  ): string {
    const header = GO_TEMPLATES.fileHeader
      .replace('{{source}}', spec.sourceFile)
      .replace('{{timestamp}}', new Date().toISOString())
      .replace('{{package}}', packageName);
    
    const imports = this.importManager.render();
    
    return `${header}
${imports}
${content}
`;
  }
  
  mapType(stdlibType: string): string {
    return mapGoType(stdlibType).type;
  }
  
  formatImports(imports: string[]): string {
    imports.forEach(i => this.importManager.add(i));
    return this.importManager.render();
  }
  
  fileHeader(spec: CodeSpec): string {
    return GO_TEMPLATES.fileHeader;
  }
  
  fileFooter(spec: CodeSpec): string {
    return '\n';  // Go files end with newline
  }
}
```
