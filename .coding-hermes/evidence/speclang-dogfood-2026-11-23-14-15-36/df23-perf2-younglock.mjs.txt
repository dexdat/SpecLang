import { execFileSync, execSync } from 'node:child_process';
import { statSync, writeFileSync, readFileSync } from 'node:fs';
const ROOT = '/home/kara/.hermes/cache/scratch/speclang-df20261123';
const BIN = `${ROOT}/bin/speclang`;
const lockPath = `${ROOT}/.speclang/cli-completion.json.lock`;
const lockBefore = statSync(lockPath);
writeFileSync(`${ROOT}/.speclang/lock-before.txt`, `size=${lockBefore.size} mtime=${lockBefore.mtime.toISOString()}\n`);
const runs = [];
for (let i = 0; i < 3; i++) {
  const t0 = Date.now();
  let rc = 0, stderrTail = '';
  try {
    execFileSync(BIN, ['validate', 'specs/examples.spec.dir/hello-world.spec.md'],

    rc = e.status ?? 'signal'; stderrTail = String(e.stderr || '').trim().split('\n')[0];
  }
  runs.push({ i, ms: Date.now() - t0, rc, stderrTail });
}
for (const r of runs) console.log(`RUN ${r.i} ms=${r.ms} rc=${r.rc} stderr='${r.stderrTail.slice(0, 120)}'`);
let lockAfter = 'present';
