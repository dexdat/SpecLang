# Fallback Protocol Engine Spec

This module implements the fallback protocols that automatically downgrade agent support levels when confidence is low, and handle peer review workflows.

## @block:types @kind:interface

### AgentSupportLevel

```typescript
export type AgentSupportLevel = 'human_only' | 'agent_assisted' | 'agent_autonomous';
```

### Spec

```typescript
export interface Spec {
  id: string;
  metadata: {
    agent_support: AgentSupportLevel;
    tags?: string[];
  };
}
```

### ConfidenceScore

```typescript
export interface ConfidenceScore {
  score: number;
  factors: {
    historicalReliability: number;
    authorReputation: number;
    similarityToGood: number;
    testCoverage: number;
    reviewStatus: number;
  };
}
```

### FallbackAction

```typescript
export interface FallbackAction {
  type: 'proceed' | 'warn' | 'review' | 'quarantine' | 'downgrade';
  spec: Spec;
  previousLevel: AgentSupportLevel;
  newLevel: AgentSupportLevel;
  reason: string;
  details: {
    confidenceScore?: number;
    threshold?: number;
    detectedIssues?: string[];
    ticketId?: string;
    reviewers?: string[];
  };
  timestamp: string;
  autoGenerated: boolean;
}
```

### FallbackConfig

```typescript
export interface FallbackConfig {
  thresholds: {
    quarantine: number;
    review: number;
    warn: number;
    requireHumanReview: number;
  };
  downgradeSteps: {
    from: AgentSupportLevel;
    to: AgentSupportLevel;
  }[];
  autoDowngrade: boolean;
  requireApproval: {
    toQuarantine: boolean;
    toReview: boolean;
    toDowngrade: boolean;
  };
  cooldownMinutes: number;
  maxQuarantineDuration: number;
}
```

## @block:class @kind:class

### FallbackProtocol

```typescript
export class FallbackProtocol {
  private config: FallbackConfig;
  private confidenceScorer: ConfidenceScorer;
  private quarantineManager: QuarantineManager;
  private peerReview: PeerReviewWorkflow;
  private notification: NotificationService;
  private cooldown: Map<string, number> = new Map();
  
  constructor(
    confidenceScorer: ConfidenceScorer,
    config: Partial<FallbackConfig> = {}
  ) {
    this.confidenceScorer = confidenceScorer;
    this.quarantineManager = new QuarantineManager();
    this.peerReview = new PeerReviewWorkflow();
    this.notification = new NotificationService();
    this.config = this.mergeConfig(config);
  }

  async evaluate(spec: Spec): Promise<FallbackAction> {
    // Check cooldown
    if (this.isInCooldown(spec.id)) {
      return this.createProceedAction(spec);
    }
    
    // Get confidence score
    const score = await this.confidenceScorer.score(spec);
    const currentLevel = spec.metadata.agent_support as AgentSupportLevel;
    
    // Determine action based on score
    let action: FallbackAction;
    
    if (score.score < this.config.thresholds.quarantine) {
      action = await this.quarantine(spec, score, currentLevel);
    } else if (score.score < this.config.thresholds.review) {
      action = await this.requestReview(spec, score, currentLevel);
    } else if (score.score < this.config.thresholds.warn) {
      action = this.warn(spec, score, currentLevel);
    } else if (score.score < this.config.thresholds.requireHumanReview && 
               currentLevel !== 'human_only') {
      action = await this.downgrade(spec, score, currentLevel);
    } else {
      action = this.createProceedAction(spec);
    }
    
    // Set cooldown if action taken
    if (action.type !== 'proceed') {
      this.setCooldown(spec.id);
    }
    
    // Send notifications
    await this.sendNotifications(action);
    
    return action;
  }

  private async quarantine(
    spec: Spec,
    score: ConfidenceScore,
    currentLevel: AgentSupportLevel
  ): Promise<FallbackAction> {
    // Downgrade to human_only
    spec.metadata.agent_support = 'human_only';
    
    // Add quarantine tag
    spec.metadata.tags = [
      ...(spec.metadata.tags || []),
      'quarantined',
      `confidence:${score.score.toFixed(2)}`,
    ];
    
    // Create review ticket
    const ticket = await this.quarantineManager.createTicket({
      specId: spec.id,
      reason: `Confidence score ${score.score.toFixed(2)} below threshold ${this.config.thresholds.quarantine}`,
      severity: 'high',
      autoGenerated: true,
    });
    
    // Find reviewers
    const reviewers = await this.findReviewers(spec);
    
    return {
      type: 'quarantine',
      spec,
      previousLevel: currentLevel,
      newLevel: 'human_only',
      reason: `Confidence score ${score.score.toFixed(2)} is below quarantine threshold ${this.config.thresholds.quarantine}`,
      details: {
        confidenceScore: score.score,
        threshold: this.config.thresholds.quarantine,
        detectedIssues: this.getDetectedIssues(score),
        ticketId: ticket.id,
        reviewers,
      },
      timestamp: new Date().toISOString(),
      autoGenerated: true,
    };
  }

  private async requestReview(
    spec: Spec,
    score: ConfidenceScore,
    currentLevel: AgentSupportLevel
  ): Promise<FallbackAction> {
    // Add review tag
    spec.metadata.tags = [
      ...(spec.metadata.tags || []),
      'needs_review',
      `confidence:${score.score.toFixed(2)}`,
    ];
    
    // Create review request
    const reviewRequest = await this.peerReview.createRequest({
      specId: spec.id,
      type: 'confidence',
      priority: score.score < this.config.thresholds.review ? 'high' : 'medium',
      assignees: await this.findReviewers(spec),
    });
    
    return {
      type: 'review',
      spec,
      previousLevel: currentLevel,
      newLevel: currentLevel, // Stay at current level pending review
      reason: `Confidence score ${score.score.toFixed(2)} requires human review`,
      details: {
        confidenceScore: score.score,
        threshold: this.config.thresholds.review,
        detectedIssues: this.getDetectedIssues(score),
        ticketId: reviewRequest.id,
        reviewers: reviewRequest.assignees,
      },
      timestamp: new Date().toISOString(),
      autoGenerated: true,
    };
  }

  private warn(
    spec: Spec,
    score: ConfidenceScore,
    currentLevel: AgentSupportLevel
  ): FallbackAction {
    // Add warning tag but don't change level
    spec.metadata.tags = [
      ...(spec.metadata.tags || []),
      'low_confidence',
      `confidence:${score.score.toFixed(2)}`,
    ];
    
    return {
      type: 'warn',
      spec,
      previousLevel: currentLevel,
      newLevel: currentLevel,
      reason: `Confidence score ${score.score.toFixed(2)} is below warning threshold`,
      details: {
        confidenceScore: score.score,
        threshold: this.config.thresholds.warn,
        detectedIssues: this.getDetectedIssues(score),
      },
      timestamp: new Date().toISOString(),
      autoGenerated: true,
    };
  }

  private async downgrade(
    spec: Spec,
    score: ConfidenceScore,
    currentLevel: AgentSupportLevel
  ): Promise<FallbackAction> {
    const newLevel = this.getDowngradeLevel(currentLevel);
    
    spec.metadata.agent_support = newLevel;
    spec.metadata.tags = [
      ...(spec.metadata.tags || []),
      'downgraded',
      `confidence:${score.score.toFixed(2)}`,
    ];
    
    return {
      type: 'downgrade',
      spec,
      previousLevel: currentLevel,
      newLevel,
      reason: `Confidence score ${score.score.toFixed(2)} requires downgrade from ${currentLevel} to ${newLevel}`,
      details: {
        confidenceScore: score.score,
        threshold: this.config.thresholds.requireHumanReview,
        detectedIssues: this.getDetectedIssues(score),
      },
      timestamp: new Date().toISOString(),
      autoGenerated: true,
    };
  }

  private createProceedAction(spec: Spec): FallbackAction {
    return {
      type: 'proceed',
      spec,
      previousLevel: spec.metadata.agent_support as AgentSupportLevel,
      newLevel: spec.metadata.agent_support as AgentSupportLevel,
      reason: 'Confidence score acceptable',
      details: {},
      timestamp: new Date().toISOString(),
      autoGenerated: false,
    };
  }

  private getDowngradeLevel(current: AgentSupportLevel): AgentSupportLevel {
    const downgradeMap: Record<AgentSupportLevel, AgentSupportLevel> = {
      'agent_autonomous': 'agent_assisted',
      'agent_assisted': 'human_only',
      'human_only': 'human_only',
    };
    
    return downgradeMap[current];
  }

  private getDetectedIssues(score: ConfidenceScore): string[] {
    const issues: string[] = [];
    
    if (score.factors.historicalReliability < 0.5) {
      issues.push('Low historical reliability');
    }
    if (score.factors.authorReputation < 0.5) {
      issues.push('Unknown or low-reputation author');
    }
    if (score.factors.similarityToGood < 0.5) {
      issues.push('Low similarity to known-good specs');
    }
    if (score.factors.testCoverage < 0.5) {
      issues.push('Low test coverage');
    }
    if (score.factors.reviewStatus < 0.5) {
      issues.push('Not reviewed by humans');
    }
    
    return issues;
  }

  private async findReviewers(spec: Spec): Promise<string[]> {
    // Find reviewers based on expertise matching spec tags
    const tags = spec.metadata.tags || [];
    
    // Simple implementation: return configured reviewers
    const allReviewers = ['reviewer1', 'reviewer2'];
    
    return allReviewers.slice(0, 2);
  }

  private async sendNotifications(action: FallbackAction): Promise<void> {
    switch (action.type) {
      case 'quarantine':
        await this.notification.send({
          type: 'quarantine',
          specId: action.spec.id,
          message: action.reason,
          ticketId: action.details.ticketId,
        });
        break;
        
      case 'review':
        await this.notification.send({
          type: 'review_request',
          specId: action.spec.id,
          message: action.reason,
          assignees: action.details.reviewers,
        });
        break;
        
      case 'downgrade':
        await this.notification.send({
          type: 'downgrade',
          specId: action.spec.id,
          message: action.reason,
          previousLevel: action.previousLevel,
          newLevel: action.newLevel,
        });
        break;
        
      case 'warn':
        await this.notification.send({
          type: 'warning',
          specId: action.spec.id,
          message: action.reason,
        });
        break;
    }
  }

  private isInCooldown(specId: string): boolean {
    const lastAction = this.cooldown.get(specId);
    if (!lastAction) return false;
    
    const cooldownMs = this.config.cooldownMinutes * 60 * 1000;
    return Date.now() - lastAction < cooldownMs;
  }

  private setCooldown(specId: string): void {
    this.cooldown.set(specId, Date.now());
  }

  private mergeConfig(config: Partial<FallbackConfig>): FallbackConfig {
    return {
      thresholds: { ...defaultConfig.thresholds, ...config.thresholds },
      downgradeSteps: config.downgradeSteps || defaultConfig.downgradeSteps,
      autoDowngrade: config.autoDowngrade ?? defaultConfig.autoDowngrade,
      requireApproval: { ...defaultConfig.requireApproval, ...config.requireApproval },
      cooldownMinutes: config.cooldownMinutes ?? defaultConfig.cooldownMinutes,
      maxQuarantineDuration: config.maxQuarantineDuration ?? defaultConfig.maxQuarantineDuration,
    };
  }

  // Manual override methods
  async approveQuarantine(specId: string, approver: string): Promise<void> {
    await this.quarantineManager.approve(specId, approver);
  }

  async releaseFromQuarantine(specId: string, approver: string): Promise<void> {
    await this.quarantineManager.release(specId, approver);
  }

  async restoreLevel(specId: string, newLevel: AgentSupportLevel): Promise<void> {
    // Restore spec to higher autonomy level
    // Should only be called after human review approves
  }
}
```
