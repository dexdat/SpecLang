# Quarantine Manager Spec

This module manages quarantine tickets for specs with low confidence scores.

## @block:types @kind:interface

### QuarantineTicket

```typescript
export interface QuarantineTicket {
  id: string;
  specId: string;
  reason: string;
  severity: 'low' | 'medium' | 'high' | 'critical';
  status: 'open' | 'approved' | 'released' | 'expired';
  createdAt: string;
  expiresAt: string;
  approvedBy: string | null;
  approvedAt: string | null;
  releaseReason: string | null;
  autoGenerated: boolean;
}
```

## @block:class @kind:class

### QuarantineManager

```typescript
export class QuarantineManager {
  private tickets: Map<string, QuarantineTicket> = new Map();
  private db: Database;
  
  constructor() {
    this.db = new Database('safety/quarantine.db');
  }

  async createTicket(data: {
    specId: string;
    reason: string;
    severity: QuarantineTicket['severity'];
    autoGenerated: boolean;
  }): Promise<QuarantineTicket> {
    const ticket: QuarantineTicket = {
      id: this.generateId(),
      specId: data.specId,
      reason: data.reason,
      severity: data.severity,
      status: 'open',
      createdAt: new Date().toISOString(),
      expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(), // 24 hours
      approvedBy: null,
      approvedAt: null,
      releaseReason: null,
      autoGenerated: data.autoGenerated,
    };
    
    this.tickets.set(ticket.id, ticket);
    
    // Persist to database
    await this.db.run(`
      INSERT INTO quarantine_tickets (id, spec_id, reason, severity, status, created_at, expires_at, auto_generated)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `, ticket.id, ticket.specId, ticket.reason, ticket.severity, ticket.status, ticket.createdAt, ticket.expiresAt, ticket.autoGenerated);
    
    return ticket;
  }

  async approve(ticketId: string, approver: string): Promise<void> {
    const ticket = this.tickets.get(ticketId);
    if (!ticket) throw new Error(`Ticket ${ticketId} not found`);
    
    ticket.status = 'approved';
    ticket.approvedBy = approver;
    ticket.approvedAt = new Date().toISOString();
    
    await this.db.run(`
      UPDATE quarantine_tickets SET status = ?, approved_by = ?, approved_at = ? WHERE id = ?
    `, ticket.status, ticket.approvedBy, ticket.approvedAt, ticket.id);
  }

  async release(ticketId: string, releaser: string, reason?: string): Promise<void> {
    const ticket = this.tickets.get(ticketId);
    if (!ticket) throw new Error(`Ticket ${ticketId} not found`);
    
    ticket.status = 'released';
    ticket.releaseReason = reason || 'Manually released';
    
    await this.db.run(`
      UPDATE quarantine_tickets SET status = ?, release_reason = ? WHERE id = ?
    `, ticket.status, ticket.releaseReason, ticket.id);
  }

  async getOpenTickets(): Promise<QuarantineTicket[]> {
    return Array.from(this.tickets.values()).filter(t => t.status === 'open');
  }

  async getTicketsForSpec(specId: string): Promise<QuarantineTicket[]> {
    return Array.from(this.tickets.values()).filter(t => t.specId === specId);
  }

  private generateId(): string {
    return `QT-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  }
}
```
